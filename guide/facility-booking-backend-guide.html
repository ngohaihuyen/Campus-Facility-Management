    description VARCHAR(255),

    category_id INT NOT NULL,

    facility_id INT NOT NULL,
    booking_date DATE NOT NULL,

4.1db.js: talking to MySQL

Every other file uses just two helpers:

how to use them
// SELECT -> array of rows
const rooms = await query('SELECT * FROM facilities WHERE id = :id', { id: 5 });

// INSERT / UPDATE -> { insertId, affectedRows }
const result = await query('UPDATE users SET is_active = 0 WHERE id = :id', { id: 5 });
if (result.affectedRows === 0) { /* nothing was changed */ }

// several statements: all saved together, or none
await transaction(async (q) => {
  await q('...');
  await q('...');
});
Always use :parameters

Never write `... WHERE email = '${email}'`. Putting user text directly into SQL makes SQL injection possible.

// src/db.js
// Connects to MySQL and gives the rest of the app two helpers:
//
//   query(sql, params)
//     SELECT         -> returns an array of rows
//     INSERT/UPDATE  -> returns { insertId, affectedRows }
//
//   transaction(async (q) => { ... })
//     Runs several queries as ONE unit: if anything fails, nothing is saved.

const mysql = require('mysql2/promise');

const pool = mysql.createPool({
  host: process.env.DB_HOST,
  port: Number(process.env.DB_PORT),
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
  namedPlaceholders: true, // write :name in SQL instead of ?
  dateStrings: true,       // dates come back as text, e.g. "2026-09-20"
});

async function query(sql, params = {}) {
  // Values are always sent separately from the SQL text. This prevents SQL injection.
  const [result] = await pool.query(sql, params);
  return result;
}

async function transaction(work) {
  const connection = await pool.getConnection();
  try {
    await connection.query('SET TRANSACTION ISOLATION LEVEL READ COMMITTED');
    await connection.beginTransaction();

    const q = async (sql, params = {}) => {
      const [result] = await connection.query(sql, params);
      return result;
    };
    const result = await work(q);

    await connection.commit();   // save everything
    return result;
  } catch (err) {
    await connection.rollback(); // undo everything
    throw err;
  } finally {
    connection.release();        // give the connection back to the pool
  }
}

module.exports = { pool, query, transaction };

// src/middleware.js
// Small functions that run before (or after) the route handlers.

const jwt = require('jsonwebtoken');
const { query } = require('./db');

// Make an error that carries an HTTP status code.
// Example: throw httpError(404, 'Facility not found.');
function httpError(status, message) {
  const err = new Error(message);
  err.status = status;
  return err;
}

// Only logged-in users can pass. Puts the user into req.user.
// The request must have the header:  Authorization: Bearer <token>
async function requireLogin(req, res, next) {
  const token = (req.headers.authorization || '').replace('Bearer ', '');
  if (!token) throw httpError(401, 'Please log in first.');

  let payload;
  try {
    payload = jwt.verify(token, process.env.JWT_SECRET);
  } catch {
    throw httpError(401, 'Your login has expired. Please log in again.');
  }

  // Read the user from the database again, so a locked account is blocked right away
  const users = await query(
    'SELECT id, name, email, role, is_active FROM users WHERE id = :id',
    { id: payload.userId },
  );
  const user = users[0];
  if (!user) throw httpError(401, 'Account not found.');
  if (!user.is_active) throw httpError(403, 'Your account is locked.');

  req.user = user; // { id, name, email, role, is_active }
  next();
}

// Only the listed roles can pass. Use it after requireLogin.
// Example: router.use(requireLogin, requireRole('ADMIN'));
function requireRole(...roles) {
  return (req, res, next) => {
    if (!roles.includes(req.user.role)) {
      throw httpError(403, 'You do not have permission to do this.');
    }
    next();
  };
}

// Every error ends up here and is sent back as JSON: { "message": "..." }
// eslint-disable-next-line no-unused-vars
function errorHandler(err, req, res, next) {
  if (err.status) {
    return res.status(err.status).json({ message: err.message });
  }
  // Common MySQL errors
  if (err.code === 'ER_DUP_ENTRY') {
    return res.status(409).json({ message: 'This value already exists.' });
  }
  if (err.code === 'ER_NO_REFERENCED_ROW' || err.code === 'ER_NO_REFERENCED_ROW_2') {
    return res.status(400).json({ message: 'An id you sent does not exist.' });
  }
  if (err.code === 'ER_CHECK_CONSTRAINT_VIOLATED' || err.code === 'WARN_DATA_TRUNCATED') {
    return res.status(400).json({ message: 'Some data is not valid.' });
  }

  console.error(err); // unexpected error: details are shown in the terminal only
  res.status(500).json({ message: 'Server error. Please try again.' });
}

module.exports = { httpError, requireLogin, requireRole, errorHandler };

function domain() {
  return process.env.EMAIL_DOMAIN || 'usth.edu.vn';

// Any university email: letters + (optional) student code + @usth.edu.vn
// Examples: anhnq2410071@usth.edu.vn (student), huydt@usth.edu.vn (teacher)
function emailPattern() {
  return new RegExp(`^[a-z]{2,}[0-9]*@${domain().replace(/\./g, '\\.')}$`);
}

// A STUDENT email must also have the student code (digits) at the end.
// anhnq2410071@usth.edu.vn  ✔      huydt@usth.edu.vn  ✘ (no code)
function studentEmailPattern() {
  return new RegExp(`^[a-z]{2,}[0-9]{4,}@${domain().replace(/\./g, '\\.')}$`);
}

function checkEmail(email) {
  if (!emailPattern().test(email)) {
    throw httpError(400, `Use your university email, for example anhnq2410071@${domain()}`);
  }
  return email;
}

// Used by the register page only
function checkStudentEmail(email) {
  if (!studentEmailPattern().test(email)) {
    throw httpError(400,
      `Only student emails can register, for example anhnq2410071@${domain()} (name + student code). `
      + 'Teacher and staff accounts are made by the admin.');
  }
  return email;
}

function checkPassword(password) {
  if (typeof password !== 'string' || password.length < 8) {
    throw httpError(400, 'Password must have at least 8 characters.');
  }
  return password;
}

A day is cut into 28 slots of 30 minutes: 07:01–07:30, 07:31–08:00, … 20:31–21:00. One booking is exactly one slot, so two bookings can never overlap by accident, and the user picks a slot from a list instead of typing a time.

// src/slots.js
// The day is cut into 30-minute slots:
//   07:01-07:30, 07:31-08:00, 08:01-08:30, ... , 20:31-21:00
// A booking is exactly ONE slot, so two bookings can never overlap.

const OPEN_HOUR = 7;    // first slot starts at 07:01
const CLOSE_HOUR = 21;  // last slot ends at 21:00
const SLOT_MINUTES = 30;

// 421 -> "07:01"
function toTime(minutesOfDay) {
  const pad = (n) => String(n).padStart(2, '0');
  return `${pad(Math.floor(minutesOfDay / 60))}:${pad(minutesOfDay % 60)}`;
}

// "07:01" -> 421
function toMinutes(time) {
  const [h, m] = time.split(':').map(Number);
  return h * 60 + m;
}

// Build the list of slots once, when the server starts
const SLOTS = [];
for (let start = OPEN_HOUR * 60; start < CLOSE_HOUR * 60; start += SLOT_MINUTES) {
  SLOTS.push({
    start: toTime(start + 1),              // 07:01
    end: toTime(start + SLOT_MINUTES),     // 07:30
  });
}

// Is this exactly one of our slots?
function isSlot(startTime, endTime) {
  return SLOTS.some((s) => s.start === startTime && s.end === endTime);
}

  maxDaysAhead: 30,       // how far ahead you can book
  minPurposeLength: 10,   // the purpose must be a real sentence
  replaceAfterHours: 24,  // a teacher can take a student's slot only if it starts more than 24h from now
};

const PRIORITY = { STUDENT: 1, STAFF: 2, TEACHER: 2 };

// May "myRole" take a slot that "otherRole" has already booked?
// Yes only if my role is higher AND the slot starts more than 24 hours from now.
function canTakeOver(myRole, otherRole, bookingDate, startTime) {
  const start = new Date(`${bookingDate}T${startTime}`);
  const limit = new Date(Date.now() + RULES.replaceAfterHours * 60 * 60 * 1000);
  return PRIORITY[myRole] > PRIORITY[otherRole] && start > limit;
}

4.5routes/bookings.js: the core

checkBookingInput() first checks everything that doesn't need the database: the date, that the time is a real slot, and the purpose. Then one transaction runs six steps:

  1. Lock the room with SELECT ... FOR UPDATE.
  2. Check that the user's role may book this room (room rule first, then room type).
  3. Find overlapping bookings.
  4. Apply the teacher priority rule, or refuse and show the taken time.
  5. Save the new booking as PENDING.
  6. Cancel any student bookings that were replaced.
the overlap check
WHERE b.facility_id  = :facilityId
  AND b.booking_date = :bookingDate
  AND b.status IN ('PENDING', 'APPROVED')
  AND b.start_time < :endTime     -- other booking starts before mine ends
  AND b.end_time   > :startTime   -- and ends after mine starts
-- 10:01-10:30 vs 10:31-11:00:  10:31 < 10:30 is false  →  no overlap (allowed)

Why the lock? Without it, two students clicking “Book” in the same second could both pass the overlap check and both be saved. FOR UPDATE makes the second request wait until the first one is done. Storing the time to the second doesn't prevent this; only the lock does. We tested it: 10 requests at once, and exactly 1 was saved.

function isRealDate(text) {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(text)) return false;
  const [y, m, d] = text.split('-').map(Number);
  const date = new Date(y, m - 1, d);
  return date.getMonth() === m - 1 && date.getDate() === d; // rejects 2026-02-30
}

// Checks everything that does not need the database. Throws a 400 error if something is wrong.
function checkBookingInput(body) {
  const facilityId = Number(body.facilityId);
  const bookingDate = String(body.bookingDate || '');
  const startTime = String(body.startTime || '');
  const endTime = String(body.endTime || '');
  const purpose = String(body.purpose || '').trim();

  if (!Number.isInteger(facilityId)) throw httpError(400, 'facilityId is required.');
  if (!isRealDate(bookingDate)) throw httpError(400, 'bookingDate must look like 2026-09-20.');

  // One booking = exactly one 30-minute slot (07:01-07:30, 07:31-08:00, ...)
  if (!isSlot(startTime, endTime)) {
    throw httpError(400, `Choose one time slot, for example ${SLOTS[0].start}-${SLOTS[0].end}.`);
  }

  const start = new Date(`${bookingDate}T${startTime}`); // local time
  const lastDay = new Date();
  lastDay.setDate(lastDay.getDate() + RULES.maxDaysAhead);
  if (start <= new Date()) throw httpError(400, 'The start time must be in the future.');
  if (start > lastDay) throw httpError(400, `You can book at most ${RULES.maxDaysAhead} days ahead.`);

  // The purpose is required (minutes 2.4). The admin checks whether it is a valid reason.
  if (purpose.length < RULES.minPurposeLength) {
    throw httpError(400, 'Please enter the purpose of the booking, e.g. "Group study for the database exam".');
  }

  return { facilityId, bookingDate, startTime, endTime, purpose };
}

// POST /api/bookings
// Body: { "facilityId": 1, "bookingDate": "2026-09-20", "startTime": "10:00",
//         "endTime": "12:00", "purpose": "Group study for the database exam" }
router.post('/', requireRole('STUDENT', 'TEACHER', 'STAFF'), async (req, res) => {
  const input = checkBookingInput(req.body);
  const user = req.user;

  // One transaction: all steps succeed together, or nothing is saved.
  const result = await transaction(async (q) => {
    // Step 1. Read the room and LOCK it (FOR UPDATE).
    // If two people book the same room at the same moment, the second one waits here
    // until the first one is finished. That is how we prevent double booking.
    const rooms = await q(
      `SELECT id, status, category_id, allowed_roles FROM facilities
       WHERE id = :facilityId AND status <> 'HIDDEN'
       FOR UPDATE`,
      input,
    );
    const room = rooms[0];
    if (!room) throw httpError(404, 'Facility not found.');
    if (room.status !== 'AVAILABLE') throw httpError(409, 'This room is not available right now.');

    // Step 2. Is this user's role allowed to book this room? (minutes 2.12)
    // The room can have its own rule; if it has none, the room type decides.
    let allowedRoles = room.allowed_roles;
    if (!allowedRoles) {
      const categories = await q(
        'SELECT allowed_roles FROM facility_categories WHERE id = :categoryId',
        { categoryId: room.category_id },
      );
      allowedRoles = categories[0].allowed_roles;
    }
    if (!allowedRoles.split(',').includes(user.role)) {
      throw httpError(403, `This room can only be booked by: ${allowedRoles.split(',').join(', ')}.`);
    }

    // Step 3. Find bookings that overlap with mine (minutes 2.2 and 2.3).
    // Overlap = the other booking starts before mine ends AND ends after mine starts.
    // We use < and > (not <= and >=), so 11:00-12:00 and 12:00-14:00 do NOT overlap.
    const clashes = await q(
      `SELECT b.id, b.start_time, b.end_time, u.role,
              TIMESTAMP(b.booking_date, b.start_time) > NOW() + INTERVAL :hours HOUR AS starts_later
       FROM bookings b
       JOIN users u ON u.id = b.user_id
       WHERE b.facility_id = :facilityId
         AND b.booking_date = :bookingDate
         AND b.status IN ('PENDING', 'APPROVED')
         AND b.start_time < :endTime
         AND b.end_time   > :startTime`,
      { ...input, hours: RULES.replaceAfterHours },
    );

    // Step 4. Priority rule (minutes 2.11).
    // I may take the slot only if EVERY clashing booking belongs to a lower role
    // and starts more than 24 hours from now. Otherwise: choose another time.
    const myPriority = PRIORITY[user.role];
    const canReplaceAll = clashes.every((c) => PRIORITY[c.role] < myPriority && c.starts_later);
    if (clashes.length > 0 && !canReplaceAll) {
      const times = clashes.map((c) => `${c.start_time.slice(0, 5)}-${c.end_time.slice(0, 5)}`);
      throw httpError(409, `This room is already booked at ${times.join(', ')}. Please choose another time.`);
    }

    // Step 5. Save my booking. It always starts as PENDING (waiting for the admin).
    const inserted = await q(
      `INSERT INTO bookings (user_id, facility_id, booking_date, start_time, end_time, purpose)
       VALUES (:userId, :facilityId, :bookingDate, :startTime, :endTime, :purpose)`,
      { ...input, userId: user.id },
    );

    // Step 6. Cancel the lower-priority bookings I replaced (only happens in the priority case)
    for (const c of clashes) {
      await q(
        `UPDATE bookings SET status = 'CANCELLED', admin_note = :note WHERE id = :id`,
        { id: c.id, note: `Replaced by a ${user.role.toLowerCase()} booking (#${inserted.insertId})` },
      );
    }

    return { id: inserted.insertId, replacedBookings: clashes.length };
  });

  res.status(201).json({
    ...result,
    status: 'PENDING',
    message: 'Booking request sent. Please wait for admin approval.',
  });
});

async function api(method, url, body) {
  const options = { method, headers: {} };

  const token = getToken();
  if (token) options.headers.Authorization = `Bearer ${token}`;

  if (body !== undefined) {
    options.headers['Content-Type'] = 'application/json';
    options.body = JSON.stringify(body);
  }

  const response = await fetch('/api' + url, options);
  const data = await response.json().catch(() => ({}));

  // Token missing or expired -> go back to the login page
  if (response.status === 401 && token) {
    logout();
  }
  if (!response.ok) {
    throw new Error(data.message || `Request failed (${response.status})`);
  }
  return data;
}

// scripts/race-test.js
// Sends 10 booking requests for the SAME room and time at the SAME moment.
// Correct result: exactly 1 request succeeds (201) and 9 are refused (409).
// Run it while the server is running:   npm run test:race

const API = 'http://localhost:3000/api';

async function login(email) {
  const res = await fetch(`${API}/auth/login`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email, password: 'Password@123' }),
  });
  const data = await res.json();
  if (!res.ok) throw new Error(`Login failed for ${email}: ${data.message}`);
  return data.token;
}

async function main() {
  const tokens = [
    await login('antt2410071@usth.edu.vn'),
    await login('binhlm2410072@usth.edu.vn'),
  ];

  // The day 5 days from today, as "YYYY-MM-DD"
  const d = new Date();
  d.setDate(d.getDate() + 5);
  const bookingDate = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;

  // Create 10 requests and send them all at once
  const requests = [];
  for (let i = 0; i < 10; i++) {
    requests.push(fetch(`${API}/bookings`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${tokens[i % 2]}` },
      body: JSON.stringify({
        facilityId: 2,
        bookingDate,
        startTime: '15:01',   // one 30-minute slot
        endTime: '15:30',
        purpose: `Race test request number ${i + 1}`,
      }),
    }));
  }
  const responses = await Promise.all(requests);

  const codes = responses.map((r) => r.status);
  console.log('Status codes:', codes.join(' '));
  const saved = codes.filter((c) => c === 201).length;
  console.log(saved === 1 ? 'PASS: only one booking was saved' : `FAIL: ${saved} bookings were saved`);
  console.log(`(Before running again, cancel or delete the booking for room 2 on ${bookingDate}.)`);
}
